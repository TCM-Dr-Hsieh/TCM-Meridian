# TCM-Meridian - Requirements
# 中醫智慧診療系統依賴套件
#
# 一鍵安裝（Windows）：雙擊 install.cmd，會建立專案內的 .venv，並依 constraints.txt 安裝下列套件的已測試版本。
# 手動安裝：pip install -r requirements.txt -c constraints.txt
#
# 注意：
# 本版本使用 NiceGUI 作為 UI 介面。Embedding 與所有 LLM 都走 OpenAI-compatible HTTP API，
# 不需要本機 GPU，也不需要 PyTorch / sentence-transformers（舊版才需要，已移除）。

# ============== 核心套件 ==============
nicegui==3.12.1                  # UI 介面
openai==1.98.0                   # OpenAI API 客戶端（用於 LLM 通訊）

# ============== 向量資料庫與檢索 ==============
chromadb==1.0.15                 # 向量資料庫
langchain==0.3.26                # LLM 應用框架（只用 Document）
langchain-community==0.3.27      # LangChain 社群整合（只用 Chroma）

# ============== 文字處理與檢索 ==============
numpy==1.26.4                    # 數值計算
requests==2.32.4                 # HTTP 請求（Embedding endpoint）

# ============== 語音功能（UI 環境這一側；語音辨識本體在獨立的 .venv-asr，見 requirements-asr.txt）==============
SoundCard>=0.4.3,<0.5            # 本機麥克風錄音
av>=14,<17                       # 遠端麥克風的重取樣、開發用音檔重播
opencc-python-reimplemented>=0.1.7,<1   # 語音辨識簡體原稿轉繁體
onnxruntime>=1.26,<2             # 說話者標記的聲紋模型（CPU）
kaldi-native-fbank>=1.22,<2      # 聲紋模型的 fbank 前處理
